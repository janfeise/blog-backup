一、常规写法的问题

js 
function process(data) {
  if (check(data) === true) {
    console.log("数据符合条件，开始处理...");
    // 一系列复杂的逻辑
    doSomething(data);
    saveResult(data);
  }
}

  • 每多一层判断，就多一层缩进。
