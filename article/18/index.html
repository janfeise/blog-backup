js 
class Student extends Person {
    university = 'University or itshon';
    #studyHours = 0;
    #course;
    static numSubjects = 10;

    constructor(fullName, birthYear, startYear, course) {
        super(fullName, birthYear);
        this.startYear = startYear;
        this.#course = course;
    }

    introduce() {
        console.log(`I study ${this.#course} at ${this.university}`);
    }

    study(h) {
        this.#makeCoffee();
        this.#studyHours += h;
    }

    #makeCoffee() {
        return 'Here is a coffee for you';
    }

    get testScore() {
        return this._testScore;
    }

    set testScore(score) {
        this._testScore = score <= 20 ? score : 0;
    }

    static printCurriculum() {
        console.log(`There are ${this.numSubjects} subjects`);
    }
}

const student = new Student('Jonas', 2020, 2037, 'Medicine');

# 1. 类中的各个字段（Fields）解释

类中的字段指的是在类的属性定义阶段就声明的成员变量，包括公有字段、私有字段、以及 startYear 这样的实例属性。

# 公有字段（public fields）

js 
university = 'University or itshon';	// 定义在构造器外，直接定义在类里面，不放置在任何函数内
  • 说明：这是一个公有实例字段，可以被类的实例直接访问或修改。
  • 访问方式：student.university ✅
  • 作用：表示学生所属的大学，所有 Student 实例都会默认属于 "University or itshon"，但也可以手动改成其他字符串。

# 私有字段（private fields）

js 
#studyHours = 0;
#course;
  • 说明：# 开头的属性是 私有字段，只能在类的内部访问，类外访问会报错。
  • 访问方式：❌ student.#studyHours 会报错
  • 作用：
    • #studyHours：记录学生学习了多少小时，初始为 0。
    • #course：存储学生所学的课程名，在构造函数中通过参数 course 初始化。

# 其他实例属性：startYear

js 
this.startYear = startYear;
  • 说明：这是在构造函数中定义的实例属性，不是用字段语法定义，但同样属于对象的属性。
  • 作用：表示学生的入学年份。
  • 访问方式：✅ student.startYear

# 字段总结：

字段名 类型 可访问性 说明
university 公有字段 ✅ 外部可访问 所属大学
#studyHours 私有字段 ❌ 外部不可访问 学习时长
#course 私有字段 ❌ 外部不可访问 所学课程
startYear 实例属性（构造器中定义） ✅ 外部可访问 入学年份

# 2. 类中的方法（Methods）解释

类中的方法包括 公有方法、私有方法、getter/setter，我们逐一分析：

# 公有方法（public methods）

# (1) introduce()

js 
introduce() {
    console.log(`I study ${this.#course} at ${this.university}`);
}

  • 作用：打印学生自我介绍，包括课程和大学名。

  • 特点：能访问私有字段 #course（私有属性只能在类的内部访问，外部是无法访问的），公有方法可以直接在实例外调用：

    js 
    student.introduce();

# (2) study(h)

js 
study(h) {
    this.#makeCoffee();
    this.#studyHours += h;
}

  • 作用：模拟学生学习操作：

    • 先调用私有方法 #makeCoffee()（做一杯咖啡）
    • 再将学习时间累加到 #studyHours

  • 访问方式：

    js 
    student.study(5); // 增加 5 小时学习时间

# 私有方法（private method）

# #makeCoffee()

js 
#makeCoffee() {
    return 'Here is a coffee for you @ ';
}

  • 作用：只能在类的内部使用，用来返回一条字符串信息，模拟制作咖啡的过程。

  • 访问限制：外部无法调用，会报错：

    js 
    student.#makeCoffee(); // ❌ 报错

# Getter 和 Setter

# get testScore()

js 
get testScore() {
    return this._testScore;
}

  • 作用：定义一个 getter，用来读取 testScore 属性值。

  • get使用方式：

    js 
    console.log(student.testScore);

# set testScore(score)

js 
set testScore(score) {
    this._testScore = score <= 20 ? score : 0;
}

  • 作用：定义一个 setter，用来设置 testScore 属性值，并对分数进行限制（大于 20 则重置为 0）。

  • set使用方式：

    js 
    student.testScore = 15;

# 方法总结：

方法名 类型 可访问性 作用
introduce() 公有方法 ✅ 打印学生课程和大学介绍
study(h) 公有方法 ✅ 增加学习时间并调用咖啡方法
#makeCoffee() 私有方法 ❌ 制作咖啡（仅内部使用）
get testScore() getter ✅ 获取测试分数
set testScore(score) setter ✅ 设置测试分数并进行合法性检查

# 3. static 属性和 static 方法

static 成员是属于类本身的，而不是实例。只能通过类名访问，不能通过对象访问。

# static 属性

js 
static numSubjects = 10;

  • 说明：定义了一个静态属性，表示总共有多少门课程。

  • 访问方式：

    js 
    console.log(Student.numSubjects); // ✅ 10，Student为类，类才能访问static 属性
console.log(student.numSubjects); // ❌ undefined，实例无法访问

# static 方法

js 
static printCurriculum() {
    console.log(`There are ${this.numSubjects} subjects`);
}

  • 说明：定义了一个静态方法，用于打印总课程数。

  • 访问方式：

    js 
    Student.printCurriculum(); // ✅ "There are 10 subjects"

  • 注意：这里的 this 指向的是类本身 Student，而不是某个实例。

# static总结：

成员名 类型 可访问性 作用 访问方式（类访问）
numSubjects 静态属性 ✅ 表示总课程数 Student.numSubjects
printCurriculum() 静态方法 ✅ 打印课程总数信息 Student.printCurriculum()

# 补充

  • 公有成员：对外暴露接口，允许类的使用者访问和操作。
  • 私有成员：实现细节隐藏，防止外部随意修改内部状态，是封装（Encapsulation）的体现。
  • 静态成员：属于类本身而非实例，适合存放“全局配置”“工具函数”等内容。
  • getter/setter：提供对属性的间接访问和校验，是数据安全性和封装性的重要手段。

# ==建议==

# 在类中如何选择使用哪个类型的字段？

属性类型 适用场景
公有字段（public field） 属性是默认值，所有实例都相同时用，==大家都有且相同==
私有字段（private field） 属性是“内部状态”，“不希望外部访问”时用
构造函数中的实例属性 以上两种除外时使用

示例：

js 
class User {
  // ✅ 公有字段（这个属性是大家公有且相同的）
  role = 'customer';

  // ✅ 私有字段（内部状态 / 不对外暴露）
  #sessionToken;
  #password;

  constructor(username, email, password) {
    // ✅ 实例属性（每个实例都不同，由外部决定）
    this.username = username; // 大家都有且不相同，保密性不高
    this.email = email;
      
    this.#password = password;

    // 内部生成私有数据
    this.#sessionToken = this.#generateToken();
  }
}

# get和set

  1. get和set无需配套使用，即，可以只使用其中一个

  2. 什么时候使用？——构造函数是直接为属性复制的，没有任何的校验，当我们需要对属性进行校验时，便可以使用set

    js 
    class Student {
  #_score = 0;

  constructor(name, score) {
    this.name = name;

    // ✅ 在构造函数中使用 setter
    this.score = score; // 实际调用的是 set score(value) 方法
  }

  // setter：对分数做校验
  set score(value) {
    if (value < 0 || value > 100) {
      console.log("❌ 分数必须在 0 ~ 100 之间，自动重置为 0");
      this.#_score = 0;
    } else {
      this.#_score = value;
    }
  }
}

// ✅ 创建实例时就会触发 set 校验
const s1 = new Student("Tom", 95);
console.log(s1.score); // 95
