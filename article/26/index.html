事件监听中的优雅传参：闭包 vs bind

阅读时间约 7 分钟 · 字数 1252

代码收录

  1. 闭包：函数可记住诞生地的数据

    js 
    const handleHover = function (opacity) {
  // 闭包, 函数记住诞生地的数据: opacity
  return function (e) {
    // 事件监听会自动传入event参数
    const hoverLink = e.target.closest('a');

    // guard line: 只有悬浮在link上时才有用
    if (!hoverLink) return;

    // 获取所有的link
    navLinks.querySelectorAll('.nav__link').forEach(link => {
      if (link !== hoverLink) link.style.opacity = opacity;
    });
  };
};

navLinks.addEventListener('mouseover', handleHover(0.5));
navLinks.addEventListener('mouseout', handleHover(1));

  2. bind：改变this指向

    js 
    const handleHover = function (e) {
  // 事件监听会自动传入event参数, 无需额外传入
  const hoverLink = e.target.closest('a');

  // guard line: 只有悬浮在link上时才有用
  if (!hoverLink) return;

  // 获取所有的link
  navLinks.querySelectorAll('.nav__link').forEach(link => {
    if (link !== hoverLink) link.style.opacity = this;
  });
};

navLinks.addEventListener('mouseover', handleHover.bind(0.5));
navLinks.addEventListener('mouseout', handleHover.bind(1));

    说明：bind改变了原this指向，原this指向：绑定事件的元素navLinks，而现在this为0.5或1
